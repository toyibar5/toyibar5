// ================================== // LEVEL 1 - MODULO CALCULATOR // ================================== function hitungModulo() { let a = Number(document.getElementById("modA").value); let m = Number(document.getElementById("modM").value); let hasil = document.getElementById("modResult"); if ( document.getElementById("modA").value.trim() === "" || document.getElementById("modM").value.trim() === "" || !Number.isSafeInteger(a) || !Number.isSafeInteger(m) ) { hasil.innerHTML = 'Masukkan bilangan bulat yang valid!'; return; } if (m <= 0) { hasil.innerHTML = 'Modulus harus positif!'; return; } let sisa = ((a % m) + m) % m; let pembagi = (a - sisa) / m; hasil.innerHTML = ` Hasil:

${a} mod ${m} = ${sisa}

Proses:
${a} = (${pembagi} × ${m}) + ${sisa}
Sisa pembagian adalah ${sisa}. `; } // ================================== // LEVEL 2 - CONGRUENCE CHECKER // ================================== function cekKongruensi() { let a = Number(document.getElementById("conA").value); let b = Number(document.getElementById("conB").value); let m = Number(document.getElementById("conM").value); let hasil = document.getElementById("conResult"); if ( document.getElementById("conA").value.trim() === "" || document.getElementById("conB").value.trim() === "" || document.getElementById("conM").value.trim() === "" || !Number.isSafeInteger(a) || !Number.isSafeInteger(b) || !Number.isSafeInteger(m) ) { hasil.innerHTML = 'Semua input harus berupa bilangan bulat!'; return; } if (m <= 0) { hasil.innerHTML = 'Modulus harus positif!'; return; } let selisih = a - b; let sisa = ((selisih % m) + m) % m; if (sisa === 0) { hasil.innerHTML = ` Status: KONGRUEN

${a} ≡ ${b} (mod ${m})

Proses:
a - b = ${a} - ${b} = ${selisih}
${selisih} mod ${m} = 0
Jadi kedua bilangan kongruen. `; } else { hasil.innerHTML = ` Status: TIDAK KONGRUEN

${a} ≢ ${b} (mod ${m})

Proses:
a - b = ${a} - ${b} = ${selisih}
${selisih} mod ${m} = ${sisa}
Jadi kedua bilangan tidak kongruen. `; } } // ================================== // LEVEL 5 - NUMBER SYSTEM CALCULATOR // ================================== function konversiBilangan() { let input = document.getElementById("numberInput").value.trim(); let basis = Number(document.getElementById("numberBase").value); let hasil = document.getElementById("numberResult"); if (input === "") { hasil.innerHTML = 'Masukkan bilangan terlebih dahulu!'; return; } try { let angka = input.toUpperCase(); let tanda = 1n; if (angka.startsWith("-")) { tanda = -1n; angka = angka.substring(1); } if (angka === "") { throw new Error("Masukkan digit yang valid."); } const digit = "0123456789ABCDEF"; let nilai = 0n; for (let karakter of angka) { let angkaDigit = digit.indexOf(karakter); if (angkaDigit < 0 || angkaDigit >= basis) { throw new Error("Digit tidak sesuai dengan basis bilangan!"); } nilai = nilai * BigInt(basis) + BigInt(angkaDigit); } nilai *= tanda; hasil.innerHTML = ` Hasil Konversi:
  • Biner: ${nilai.toString(2)}
  • Desimal: ${nilai.toString(10)}
  • Oktal: ${nilai.toString(8)}
  • Heksadesimal: ${nilai.toString(16).toUpperCase()}
`; } catch (error) { hasil.innerHTML = ` ${error.message} `; } } // ================================== // LEVEL 6 - COMBINATORICS CALCULATOR // ================================== function faktorial(n) { let hasil = 1n; for (let i = 1; i <= n; i++) { hasil *= BigInt(i); } return hasil; } function hitungKombinatorika() { let n = Number(document.getElementById("combN").value); let r = Number(document.getElementById("combR").value); let hasil = document.getElementById("combResult"); if ( document.getElementById("combN").value.trim() === "" || document.getElementById("combR").value.trim() === "" || !Number.isSafeInteger(n) || !Number.isSafeInteger(r) ) { hasil.innerHTML = 'Masukkan bilangan bulat yang valid!'; return; } if (n < 0 || r < 0 || r > n) { hasil.innerHTML = ` Syarat harus terpenuhi: 0 ≤ r ≤ n. `; return; } if (n > 500) { hasil.innerHTML = 'Nilai n maksimal 500.'; return; } // Menghitung permutasi let permutasi = faktorial(n) / faktorial(n - r); // Menghitung kombinasi let kombinasi = faktorial(n) / (faktorial(r) * faktorial(n - r)); hasil.innerHTML = ` PERMUTASI

P(${n},${r}) = ${permutasi}

Rumus:
P(n,r) = n! / (n-r)!

KOMBINASI

C(${n},${r}) = ${kombinasi}

Rumus:
C(n,r) = n! / (r!(n-r)!)

Permutasi memperhatikan urutan, sedangkan kombinasi tidak memperhatikan urutan. `; }
